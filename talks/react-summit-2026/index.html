A Backtick app.

server/App.tsxSimplified 
import { cs } from "@backtickjs/core";
import { Text } from "react-native" with { backtick: "client" };
import { useState } from "react" with { backtick: "client" };
const Counter = cs`({ from }: { from: number }) => {
  const [count, setCount] = $useState(from);
  const onPress = () => setCount(count + 1);
  return <$Text onPress={onPress}>{count}</$Text>;
}`;
app.get("/:from", (c) => {
  const from = Number(c.req.param("from"));
  return c.render(cs`<$Counter from={$from} />`);
});
